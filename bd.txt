-- (Does NOT delete existing data)

-- 1. Employees Table (Create only if it doesn't exist)
CREATE TABLE IF NOT EXISTS employees (
    id SERIAL PRIMARY KEY,
    employee_number VARCHAR(20) UNIQUE NOT NULL, -- Ex: ML12345678 or MASTER01
    name VARCHAR(100) NOT NULL,
    password VARCHAR(255) NOT NULL, -- Stores hashed password
    role_id INT NOT NULL,           -- Hierarchy level (1-6)
    shift INT                       -- 1, 2, or 3
);

-- 2. Vacation Requests Table (Create only if it doesn't exist)
CREATE TABLE IF NOT EXISTS vacation_requests (
    id SERIAL PRIMARY KEY,
    employee_id INT REFERENCES employees(id), -- Who is requesting
    
    -- Request Details
    date_start DATE NOT NULL,
    date_end DATE NOT NULL,
    submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) DEFAULT 'Pending Engineer',
    
    -- Comments
    tech_comment TEXT,
    clerk_comment TEXT,
    
    -- Approval Signatures (Chain of Command ids)
    engineer_id INT REFERENCES employees(id),
    supervisor_id INT REFERENCES employees(id),
    admin_id INT REFERENCES employees(id),
    clerk_id INT REFERENCES employees(id),
    
    -- Soft Delete (History Management)
    is_hidden BOOLEAN DEFAULT FALSE
);

-- 3. Safely Insert Master User
-- "ON CONFLICT DO NOTHING" ensures that if MASTER01 already exists,
-- it won't crash or duplicate.
