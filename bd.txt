-- ============================================================
-- DATABASE SCHEMA - VACATION REQUEST SYSTEM
-- ============================================================

-- 1. Cleanup (Drop tables if they exist)
DROP TABLE IF EXISTS vacation_requests CASCADE;
DROP TABLE IF EXISTS employees CASCADE;

-- 2. Employees Table
-- role_id defines the hierarchy (see below)
CREATE TABLE employees (
    id SERIAL PRIMARY KEY,
    employee_number VARCHAR(10) UNIQUE NOT NULL, -- Ex: ML12345678
    name VARCHAR(100) NOT NULL,
    password VARCHAR(255) NOT NULL, -- Stores hashed password
    role_id INT NOT NULL,           -- 1, 2, 3, 4, or 5
    shift INT                       -- 1, 2, or 3 (Can be NULL for Admin/Clerk)
);

-- 3. Vacation Requests Table
CREATE TABLE vacation_requests (
    id SERIAL PRIMARY KEY,
    employee_id INT REFERENCES employees(id), -- Who is requesting
    
    -- Request Details
    date_start DATE NOT NULL,
    date_end DATE NOT NULL,
    submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) DEFAULT 'Pending Engineer',
    
    -- Comments
    tech_comment TEXT,
    clerk_comment TEXT,
    
    -- Approval Signatures (Chain of Command)
    engineer_id INT REFERENCES employees(id),
    supervisor_id INT REFERENCES employees(id),
    admin_id INT REFERENCES employees(id),
    clerk_id INT REFERENCES employees(id),
    
    -- Soft Delete (Recycle Bin)
    is_hidden BOOLEAN DEFAULT FALSE
);
